% HMAC 结构图 —— 双层哈希流向
\begin{figure}[H]
\centering
\begin{tikzpicture}[
  node distance = 0.5cm and 0.8cm,
  box/.style    = {rectangle, draw, rounded corners=3pt, minimum width=2.0cm, minimum height=0.65cm,
                   font=\small, fill=white},
  op/.style     = {circle, draw, minimum size=0.55cm, font=\small\bfseries, fill=gray!15},
  arr/.style    = {-Stealth, thick},
  lbl/.style    = {font=\small\itshape},
  grn/.style    = {fill=green!12, draw=green!50!black},
  blu/.style    = {fill=blue!10, draw=blue!50!black},
  orn/.style    = {fill=orange!18, draw=orange!70!black},
]

% 原始密钥 K
\node[box, blu] (K) {密钥 $K$};

% 密钥预处理
\node[box, below=0.7cm of K, blu] (Kp) {$K'$（填充/截断至分组长度）};

% ipad / opad
\node[box, left=1.4cm of Kp, orn]  (ipad) {$\mathit{ipad}$};
\node[box, right=1.4cm of Kp, orn] (opad) {$\mathit{opad}$};

% XOR nodes
\node[op, below=0.6cm of ipad] (xori) {$\oplus$};
\node[op, below=0.6cm of opad] (xoro) {$\oplus$};

% 消息 m
\node[box, left=0.9cm of xori, blu] (msg) {消息 $m$};

% 内层哈希输入拼接
\node[box, grn, below=0.6cm of xori, minimum width=3.4cm] (inner_in) {$(K' \oplus \mathit{ipad}) \;\|\; m$};

% 内层哈希
\node[box, grn, below=0.5cm of inner_in] (H1) {$H(\cdot)$（内层哈希）};

% 外层拼接：xoro 结果 || H1 结果
\node[box, orn, below=0.7cm of H1, minimum width=3.4cm] (outer_in) {$(K' \oplus \mathit{opad}) \;\|\; H(\text{inner})$};

% 外层哈希
\node[box, orn, below=0.5cm of outer_in] (H2) {$H(\cdot)$（外层哈希）};

% 输出
\node[box, below=0.5cm of H2, minimum width=2.8cm, fill=red!10, draw=red!50!black] (out) {$\operatorname{HMAC}(K, m)$};

% 箭头
\draw[arr] (K)  -- (Kp);
\draw[arr] (Kp) -- (ipad);
\draw[arr] (Kp) -- (opad);
\draw[arr] (Kp) -- (xori);
\draw[arr] (Kp) -- (xoro);
\draw[arr] (ipad) -- (xori);
\draw[arr] (opad) -- (xoro);
\draw[arr] (msg)  -- (xori);
\draw[arr] (xori) -- (inner_in);
\draw[arr] (inner_in) -- (H1);
\draw[arr] (H1) -- (outer_in);
\draw[arr] (xoro.south) -- ++(0,-0.3) -| (outer_in.north);
\draw[arr] (outer_in) -- (H2);
\draw[arr] (H2) -- (out);

% 注释
\node[lbl, right=0.2cm of xori] {内层混合};
\node[lbl, right=0.2cm of H2]   {外层混合};

\end{tikzpicture}
\caption{HMAC 双层嵌套哈希结构}
\label{fig:hmac_structure}
\end{figure}
